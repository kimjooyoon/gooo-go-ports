const compilerRevision = "83928fa4538245abe33b08549da47f8cbdc50ed2"
const recordID = "gooo://ipv4-address-facts/result/v1"
const maxJSONBytes = 32 << 20

var fieldNames = []string{"a", "b", "c", "d", "valid", "is_private", "is_loopback", "is_unspecified", "is_multicast", "is_link_local_unicast", "is_global_unicast"}

type fileRef struct {
	Path   string `json:"path"`
	Bytes  int    `json:"bytes"`
	SHA256 string `json:"sha256"`
}

func digest(raw []byte) string             { return fmt.Sprintf("sha256:%x", sha256.Sum256(raw)) }
func rawDigest(raw []byte) string          { return fmt.Sprintf("%x", sha256.Sum256(raw)) }
func namedDigest(prefix, id string) string { return prefix + rawDigest([]byte(id)) }

// This scans every subtree, including opaque values consumed by the unchanged
// typed Result serializer. Numbers remain json.Number, never float64.
func decode(raw []byte) (any, error) {
	if len(raw) == 0 || len(raw) > maxJSONBytes || !utf8.Valid(raw) {
		return nil, fmt.Errorf("invalid JSON size or raw UTF-8")
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.UseNumber()
	value, err := jsonValue(d, 0)
	if err != nil {
		return nil, err
	}
	if _, err := d.Token(); err != io.EOF {
		return nil, fmt.Errorf("trailing JSON")
	}
	return value, nil
}

func jsonValue(d *json.Decoder, depth int) (any, error) {
	if depth > 64 {
		return nil, fmt.Errorf("JSON exceeds64 nested levels")
	}
	t, err := d.Token()
	if err != nil {
		return nil, err
	}
	delim, container := t.(json.Delim)
	if !container {
		return t, nil
	}
	switch delim {
	case '{':
		out := make(map[string]any)
		for d.More() {
			t, err := d.Token()
			if err != nil {
				return nil, err
			}
			key, ok := t.(string)
			if !ok {
				return nil, fmt.Errorf("object key required")
			}
			if _, exists := out[key]; exists {
				return nil, fmt.Errorf("duplicate JSON key %q", key)
			}
			value, err := jsonValue(d, depth+1)
			if err != nil {
				return nil, err
			}
			out[key] = value
		}
		close, err := d.Token()
		if err != nil {
			return nil, err
		}
		if close != json.Delim('}') {
			return nil, fmt.Errorf("object close required")
		}
		return out, nil
	case '[':
		out := make([]any, 0)
		for d.More() {
			v, err := jsonValue(d, depth+1)
			if err != nil {
				return nil, err
			}
			out = append(out, v)
		}
		close, err := d.Token()
		if err != nil {
			return nil, err
		}
		if close != json.Delim(']') {
			return nil, fmt.Errorf("array close required")
		}
		return out, nil
	default:
		return nil, fmt.Errorf("unexpected delimiter")
	}
}

func readFile(path string, cap int) ([]byte, error) {
	if !filepath.IsAbs(path) || filepath.Clean(path) != path {
		return nil, fmt.Errorf("canonical absolute path required")
	}
	resolved, err := filepath.EvalSymlinks(path)
	if err != nil || resolved != path {
		return nil, fmt.Errorf("symlink or noncanonical path")
	}
	before, err := os.Lstat(path)
	if err != nil {
		return nil, err
	}
	if !before.Mode().IsRegular() || before.Size() <= 0 || before.Size() > int64(cap) {
		return nil, fmt.Errorf("bounded regular file required")
	}
	f, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer f.Close()
	info, err := f.Stat()
	if err != nil || !os.SameFile(before, info) {
		return nil, fmt.Errorf("file changed before read")
	}
	raw, err := io.ReadAll(io.LimitReader(f, int64(cap)+1))
	if err != nil {
		return nil, err
	}
	after, err := os.Lstat(path)
	if err != nil || !after.Mode().IsRegular() || !os.SameFile(before, after) || before.Size() != after.Size() ||
		after.Size() != int64(len(raw)) || len(raw) > cap || !before.ModTime().Equal(after.ModTime()) {
		return nil, fmt.Errorf("file changed during read")
	}
	return raw, nil
}

func currentRef(path string, cap int) (fileRef, error) {
	raw, err := readFile(path, cap)
	if err != nil {
		return fileRef{}, err
	}
	return fileRef{path, len(raw), rawDigest(raw)}, nil
}

func checkRef(ref fileRef, cap int) error {
	actual, err := currentRef(ref.Path, cap)
	if err != nil {
		return err
	}
	if actual != ref {
		return fmt.Errorf("file reference differs: %s", ref.Path)
	}
	return nil
}

func exactDecode(raw []byte, out any) error {
	if _, err := decode(raw); err != nil {
		return err
	}
	d := json.NewDecoder(bytes.NewReader(raw))
	d.DisallowUnknownFields()
	return d.Decode(out)
}

func nameIs(expr ast.Expr, name string) bool { v, ok := expr.(*ast.Ident); return ok && v.Name == name }

// Validate before compiling the candidate into the harness. Source bytes are
// never reformatted. Provenance/body equality is additionally checked by the
// Python receipt observer; this is a concrete, effect-free compiler839 ABI.
func validateGenerated(raw []byte) error {
	if len(raw) == 0 || len(raw) > 1<<20 || !utf8.Valid(raw) {
		return fmt.Errorf("generated source bound")
	}
	f, err := parser.ParseFile(token.NewFileSet(), "generated.go", raw, parser.ParseComments)
	if err != nil {
		return err
	}
	if f.Name.Name != "main" || len(f.Imports) != 0 {
		return fmt.Errorf("generated package/import ABI")
	}
	for _, group := range f.Comments {
		for _, c := range group.List {
			text := strings.TrimSpace(c.Text)
			if strings.HasPrefix(text, "//go:") || constraint.IsPlusBuild(text) {
				return fmt.Errorf("generated directive forbidden")
			}
		}
	}
	recordName := namedDigest("GoooRecord", recordID)
	funcs := make(map[string]*ast.FuncDecl)
	records := 0
	for _, declaration := range f.Decls {
		switch d := declaration.(type) {
		case *ast.GenDecl:
			if d.Tok != token.TYPE || len(d.Specs) != 1 {
				return fmt.Errorf("generated globals forbidden")
			}
			spec, ok := d.Specs[0].(*ast.TypeSpec)
			if !ok || spec.Assign.IsValid() || spec.Name.Name != recordName {
				return fmt.Errorf("generated record identity")
			}
			st, ok := spec.Type.(*ast.StructType)
			if !ok || len(st.Fields.List) != len(fieldNames) {
				return fmt.Errorf("generated record fields")
			}
			for i, field := range st.Fields.List {
				kind := "bool"
				if i < 4 {
					kind = "int64"
				}
				tag := ""
				if field.Tag != nil {
					tag, _ = strconv.Unquote(field.Tag.Value)
				}
				if len(field.Names) != 1 || field.Names[0].Name != namedDigest("GoooField", recordID+"/"+fieldNames[i]) ||
					!nameIs(field.Type, kind) || tag != "json:\""+fieldNames[i]+"\"" {
					return fmt.Errorf("generated field identity/type/tag/order")
				}
			}
			records++
		case *ast.FuncDecl:
			if funcs[d.Name.Name] != nil || d.Recv != nil || d.Type.TypeParams != nil {
				return fmt.Errorf("generated function identity")
			}
			funcs[d.Name.Name] = d
		default:
			return fmt.Errorf("unexpected generated declaration")
		}
	}
	if records != 1 || len(funcs) != 8 {
		return fmt.Errorf("generated function/record count")
	}
	for i := -1; i < 7; i++ {
		name := "GoooComposedActivity0"
		if i >= 0 {
			name += fmt.Sprint("Call", i)
		}
		fn := funcs[name]
		if fn == nil || fn.Body == nil || len(fn.Type.Params.List) != 4 || fn.Type.Results == nil || len(fn.Type.Results.List) != 1 {
			return fmt.Errorf("generated entry/callee signature")
		}
		for j, param := range fn.Type.Params.List {
			if len(param.Names) != 1 || param.Names[0].Name != fmt.Sprint("input", j) || !nameIs(param.Type, "int64") {
				return fmt.Errorf("generated four-int64 ABI")
			}
		}
		result := "bool"
		if i == -1 {
			result = recordName
		}
		if len(fn.Type.Results.List[0].Names) != 0 || !nameIs(fn.Type.Results.List[0].Type, result) {
			return fmt.Errorf("generated return ABI")
		}
		calls := make(map[string]int)
		var invalid error
		ast.Inspect(fn.Body, func(n ast.Node) bool {
			if n == nil || invalid != nil {
				return invalid == nil
			}
			switch node := n.(type) {
			case *ast.BlockStmt, *ast.ReturnStmt, *ast.IfStmt, *ast.ParenExpr, *ast.Ident,
				*ast.BasicLit, *ast.CompositeLit, *ast.KeyValueExpr, *ast.DeclStmt, *ast.ValueSpec:
			case *ast.GenDecl:
				if i != -1 || node.Tok != token.VAR {
					invalid = fmt.Errorf("generated local declaration")
				}
			case *ast.UnaryExpr:
				if node.Op != token.NOT {
					invalid = fmt.Errorf("generated unary effect")
				}
			case *ast.BinaryExpr:
				switch node.Op {
				case token.LAND, token.LOR, token.EQL, token.NEQ, token.LSS, token.LEQ, token.GTR, token.GEQ:
				default:
					invalid = fmt.Errorf("generated binary effect")
				}
			case *ast.CallExpr:
				target, ok := node.Fun.(*ast.Ident)
				if !ok || node.Ellipsis.IsValid() || len(node.Args) != 4 {
					invalid = fmt.Errorf("generated pure-call ABI")
					break
				}
				for j, arg := range node.Args {
					if !nameIs(arg, fmt.Sprint("input", j)) {
						invalid = fmt.Errorf("generated pure-call arguments")
					}
				}
				calls[target.Name]++
			default:
				invalid = fmt.Errorf("generated effect or unsupported node %T", node)
			}
			return invalid == nil
		})
		if invalid != nil {
			return invalid
		}
		expected := make(map[string]int)
		if i == -1 {
			for j := 0; j < 7; j++ {
				expected[fmt.Sprint("GoooComposedActivity0Call", j)] = 1
			}
		}
		if i > 0 {
			expected["GoooComposedActivity0Call0"] = 1
		}
		if i == 6 {
			for _, j := range []int{2, 3, 4, 5} {
				expected[fmt.Sprint("GoooComposedActivity0Call", j)] = 1
			}
		}
		if len(calls) != len(expected) {
			return fmt.Errorf("generated call graph")
		}
		for name, n := range expected {
			if calls[name] != n {
				return fmt.Errorf("generated call target/count")
			}
		}
	}
	return nil
}

// The optional pair ABI is a second closed compiler839 shape. It does not
// relax the classifier's four-int64 validator above.
const pairRuleID = "gooo://ipv4-rule-pair/rule/v1"
const pairResultID = "gooo://ipv4-rule-pair/result/v2"

var pairNames = []string{"valid", "comparable", "overlap", "opposing_effect", "conflict", "witness_valid", "witness_a", "witness_b", "witness_c", "witness_d", "reason"}
var pairRuleNames = []string{"a", "b", "c", "d", "bits", "effect", "scope", "priority"}

func validatePairGenerated(raw []byte) error {
	if len(raw) == 0 || len(raw) > 1<<20 || !utf8.Valid(raw) {
		return fmt.Errorf("pair source bound")
	}
	f, err := parser.ParseFile(token.NewFileSet(), "generated.go", raw, parser.ParseComments)
	if err != nil {
		return err
	}
	if f.Name.Name != "main" || len(f.Imports) != 0 {
		return fmt.Errorf("pair package/import ABI")
	}
	for _, g := range f.Comments {
		for _, c := range g.List {
			s := strings.TrimSpace(c.Text)
			if strings.HasPrefix(s, "//go:") || constraint.IsPlusBuild(s) {
				return fmt.Errorf("pair directive")
			}
		}
	}
	address := namedDigest("GoooRecord", "gooo://go/net/netip/ipv4/address/v1")
	prefix := namedDigest("GoooRecord", "gooo://go/net/netip/ipv4/prefix/v1")
	rule := namedDigest("GoooRecord", pairRuleID)
	result := namedDigest("GoooRecord", pairResultID)
	ids := map[string]string{address: "gooo://go/net/netip/ipv4/address/v1", prefix: "gooo://go/net/netip/ipv4/prefix/v1", rule: pairRuleID, result: pairResultID}
	fields := map[string][]string{address: {"a", "b", "c", "d"}, prefix: {"a", "b", "c", "d", "bits"}, rule: pairRuleNames, result: pairNames}
	found := map[string]bool{}
	funcs := map[string]*ast.FuncDecl{}
	selectors := map[string]bool{}
	for _, d := range f.Decls {
		switch d := d.(type) {
		case *ast.GenDecl:
			if d.Tok != token.TYPE || len(d.Specs) != 1 {
				return fmt.Errorf("pair global declaration")
			}
			t, ok := d.Specs[0].(*ast.TypeSpec)
			if !ok || t.Assign.IsValid() || ids[t.Name.Name] == "" || found[t.Name.Name] {
				return fmt.Errorf("pair record identity")
			}
			st, ok := t.Type.(*ast.StructType)
			if !ok || len(st.Fields.List) != len(fields[t.Name.Name]) {
				return fmt.Errorf("pair record fields")
			}
			for i, x := range st.Fields.List {
				name := fields[t.Name.Name][i]
				kind := "int64"
				if t.Name.Name == result && i < 6 {
					kind = "bool"
				}
				tag := ""
				if x.Tag != nil {
					tag, _ = strconv.Unquote(x.Tag.Value)
				}
				goName := namedDigest("GoooField", ids[t.Name.Name]+"/"+name)
				if len(x.Names) != 1 || x.Names[0].Name != goName || !nameIs(x.Type, kind) || tag != "json:\""+name+"\"" {
					return fmt.Errorf("pair field ID/type/tag/order")
				}
				selectors[goName] = true
			}
			found[t.Name.Name] = true
		case *ast.FuncDecl:
			if d.Recv != nil || d.Type.TypeParams != nil || funcs[d.Name.Name] != nil {
				return fmt.Errorf("pair function identity")
			}
			funcs[d.Name.Name] = d
		default:
			return fmt.Errorf("pair declaration forbidden")
		}
	}
	if len(found) != 4 || len(funcs) != 5 {
		return fmt.Errorf("pair record/function count")
	}
	signatures := map[string][]string{"GoooComposedActivity0": {rule, rule, result}, "GoooComposedActivity0Call0": {rule, "bool"}, "GoooComposedActivity0Call1": {prefix, "bool"}, "GoooComposedActivity0Call2": {"int64", "int64"}, "GoooComposedActivity0Call3": {prefix, prefix, "bool"}}
	expected := map[string]map[string]int{"GoooComposedActivity0": {"GoooComposedActivity0Call0": 2, "GoooComposedActivity0Call2": 1, "GoooComposedActivity0Call3": 1}, "GoooComposedActivity0Call0": {}, "GoooComposedActivity0Call1": {}, "GoooComposedActivity0Call2": {}, "GoooComposedActivity0Call3": {"GoooComposedActivity0Call1": 2, "GoooComposedActivity0Call2": 1}}
	for name, signature := range signatures {
		fn := funcs[name]
		n := len(signature) - 1
		if fn == nil || fn.Body == nil || fn.Type.Params == nil || len(fn.Type.Params.List) != n || fn.Type.Results == nil || len(fn.Type.Results.List) != 1 || len(fn.Type.Results.List[0].Names) != 0 || !nameIs(fn.Type.Results.List[0].Type, signature[n]) {
			return fmt.Errorf("pair function signature")
		}
		for i, p := range fn.Type.Params.List {
			want := "input"
			if n > 1 {
				want += strconv.Itoa(i)
			}
			if len(p.Names) != 1 || p.Names[0].Name != want || !nameIs(p.Type, signature[i]) {
				return fmt.Errorf("pair parameter ABI")
			}
		}
		calls := map[string]int{}
		var bad error
		ast.Inspect(fn.Body, func(node ast.Node) bool {
			if node == nil || bad != nil {
				return bad == nil
			}
			switch v := node.(type) {
			case *ast.BlockStmt, *ast.ReturnStmt, *ast.IfStmt, *ast.ParenExpr, *ast.Ident, *ast.BasicLit, *ast.CompositeLit, *ast.KeyValueExpr, *ast.DeclStmt, *ast.ValueSpec:
			case *ast.GenDecl:
				if v.Tok != token.VAR {
					bad = fmt.Errorf("pair local declaration")
				}
			case *ast.SelectorExpr:
				if !selectors[v.Sel.Name] {
					bad = fmt.Errorf("pair field selector")
				}
			case *ast.AssignStmt:
				if v.Tok != token.ASSIGN || len(v.Lhs) != 1 || len(v.Rhs) != 1 {
					bad = fmt.Errorf("pair assignment")
				}
				for _, x := range v.Lhs {
					if _, ok := x.(*ast.Ident); !ok {
						bad = fmt.Errorf("pair assignment target")
					}
				}
			case *ast.UnaryExpr:
				if v.Op != token.NOT && v.Op != token.SUB {
					bad = fmt.Errorf("pair unary effect")
				}
			case *ast.BinaryExpr:
				switch v.Op {
				case token.LAND, token.LOR, token.EQL, token.NEQ, token.LSS, token.LEQ, token.GTR, token.GEQ, token.ADD, token.SUB, token.MUL, token.QUO, token.REM:
				default:
					bad = fmt.Errorf("pair binary effect")
				}
			case *ast.CallExpr:
				t, ok := v.Fun.(*ast.Ident)
				if !ok || v.Ellipsis.IsValid() {
					bad = fmt.Errorf("pair call target")
					break
				}
				sig, ok := signatures[t.Name]
				if !ok || len(v.Args) != len(sig)-1 || expected[name][t.Name] == 0 {
					bad = fmt.Errorf("pair call ABI")
					break
				}
				calls[t.Name]++
			default:
				bad = fmt.Errorf("pair effect or unsupported node %T", node)
			}
			return bad == nil
		})
		if bad != nil {
			return bad
		}
		if len(calls) != len(expected[name]) {
			return fmt.Errorf("pair call roster")
		}
		for callee, count := range expected[name] {
			if calls[callee] != count {
				return fmt.Errorf("pair call count")
			}
		}
	}
	return nil
}
