package main

import (
	"encoding/json"
	"flag"
	"fmt"
	"io"
	"os"
	"runtime"
)

// The unchanged public codec/profile are copied into this owned module before
// build. The bridge never contains candidate generated code or a policy oracle.
func runBridge(args []string) error {
	fs := flag.NewFlagSet("ipv4-digest-and-ABI", flag.ContinueOnError)
	fs.SetOutput(io.Discard)
	receipt := fs.String("receipt", "", "canonical original execute receipt")
	generated := fs.String("check-generated", "", "canonical generated source to inspect before compile")
	if err := fs.Parse(args); err != nil {
		return err
	}
	if fs.NArg() != 0 || (*receipt == "") == (*generated == "") {
		return fmt.Errorf("require exactly one --receipt or --check-generated")
	}
	if runtime.Version() != "go1.27.2" {
		return fmt.Errorf("pinned Go version required")
	}
	if len(resultProfileJSON) != 58326 || rawDigest(resultProfileJSON) != "9eafadab81a6c7fe33a4f59777e3776cc27f31c7962e472f3828e77867142d44" {
		return fmt.Errorf("embedded result profile differs")
	}
	if *generated != "" {
		raw, err := readFile(*generated, 1<<20)
		if err != nil {
			return err
		}
		if err := validateGenerated(raw); err != nil {
			return err
		}
		ref, err := currentRef(*generated, 1<<20)
		if err != nil {
			return err
		}
		return json.NewEncoder(os.Stdout).Encode(map[string]any{"schema": "gooo/ipv4-generated-ABI/v1", "generated_source": ref, "functions": 8, "record_fields": 11, "input_ports": 4})
	}
	raw, err := readFile(*receipt, maxJSONBytes)
	if err != nil {
		return err
	}
	typed, err := pinnedResultDigest(raw)
	if err != nil {
		return err
	}
	ref, err := currentRef(*receipt, maxJSONBytes)
	if err != nil {
		return err
	}
	if ref.SHA256 != rawDigest(raw) {
		return fmt.Errorf("receipt changed after digest")
	}
	return json.NewEncoder(os.Stdout).Encode(map[string]any{"schema": "gooo/ipv4-pinned-result-digest/v1", "receipt": ref, "typed_result_sha256": typed, "compiler_revision": compilerRevision})
}

func main() {
	if err := runBridge(os.Args[1:]); err != nil {
		fmt.Fprintln(os.Stderr, "IPV4_BRIDGE_REFUSED:", err)
		os.Exit(1)
	}
}
