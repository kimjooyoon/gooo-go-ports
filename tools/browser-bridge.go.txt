package main

import (
	"syscall/js"

	"github.com/kimjooyoon/gooo-go-ports/internal/demobridge"
)

func main() {
	compute := js.FuncOf(func(_ js.Value, args []js.Value) any {
		if len(args) < 1 || len(args) > 5 {
			return `{"ok":false,"error":"Expected an operation and up to four decimal strings"}`
		}
		for _, arg := range args {
			if arg.Type() != js.TypeString {
				return `{"ok":false,"error":"All arguments must be strings"}`
			}
		}
		inputs := make([]string, len(args)-1)
		for i := range inputs {
			inputs[i] = args[i+1].String()
		}
		return demobridge.Run(args[0].String(), inputs,
			func(op string, a, b, c, d int64) any { return GoooComposedActivity0(op, a, b, c, d) })
	})
	js.Global().Set("goooRun", compute)
	js.Global().Set("goooReady", true)
	select {}
}
