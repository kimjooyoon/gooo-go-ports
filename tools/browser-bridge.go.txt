package main

import (
	"syscall/js"
	"github.com/kimjooyoon/gooo-go-ports/internal/demobridge"
)

func main() {
	compute := js.FuncOf(func(_ js.Value, args []js.Value) any {
		if len(args) != 3 { return `{"ok":false,"error":"Expected operation and two decimal strings"}` }
		for _, arg := range args {
			if arg.Type() != js.TypeString { return `{"ok":false,"error":"All arguments must be strings"}` }
		}
		return demobridge.Run(args[0].String(), args[1].String(), args[2].String(),
			func(op string, a, b int64) any { return GoooComposedActivity0(op, a, b) })
	})
	js.Global().Set("goooRun", compute)
	js.Global().Set("goooReady", true)
	select {}
}
