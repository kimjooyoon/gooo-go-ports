package main

import (
    "encoding/json"
    "os"
    "runtime"
    "testing"
    "unicode"
)

// Only the oracle and reporting live here. The function under test is called
// directly from the byte-for-byte copy of the checked generated composition.
func TestSimpleFoldExhaustive(t *testing.T) {
    if runtime.Version() != "go1.27.2" || unicode.Version != "17.0.0" {
        t.Fatal("wrong pinned Go / Unicode oracle")
    }
    unicodeInputs := 0
    for x := int64(0); x <= int64(unicode.MaxRune); x++ {
        want := int64(unicode.SimpleFold(rune(x)))
        if got := {{.Function}}(x); got != want {
            t.Fatalf("SimpleFold(%d) = %d, want %d", x, got, want)
        }
        unicodeInputs++
    }
    outside := []int64{ {{range .Outside}}{{.}},{{end}} }
    outsideComparisons := 0
    for _, x := range outside {
        if x < -2147483648 || x > 2147483647 || (x >= 0 && x <= unicode.MaxRune) {
            t.Fatalf("invalid original int32 reference input %d", x)
        }
        want := int64(unicode.SimpleFold(rune(x)))
        if got := {{.Function}}(x); got != want {
            t.Fatalf("original int32 SimpleFold(%d) = %d, want %d", x, got, want)
        }
        outsideComparisons++
    }
    extensions := []int64{ {{range .Extensions}}{{.}},{{end}} }
    extensionComparisons := 0
    for _, x := range extensions {
        if x >= -2147483648 && x <= 2147483647 {
            t.Fatalf("extension input is in original rune domain: %d", x)
        }
        // This explicit identity extension must never narrow x to Go rune.
        if got := {{.Function}}(x); got != x {
            t.Fatalf("int64 extension SimpleFold(%d) = %d, want original input", x, got)
        }
        extensionComparisons++
    }
    report := map[string]any{
        "schema": "gooo/simple-fold-exhaustive-native/v1",
        "go_version": runtime.Version(), "unicode_version": unicode.Version,
        "generated_sha256": "{{.Generated}}",
        "unicode_range_inputs": unicodeInputs, "unicode_range_comparisons": unicodeInputs,
        "outside_unicode_int32_inputs": outside, "outside_unicode_int32_comparisons": outsideComparisons,
        "outside_int32_extension_inputs": extensions, "outside_int32_extension_comparisons": extensionComparisons,
        "mismatches": 0,
    }
    path := os.Getenv("GOOO_SIMPLE_FOLD_REPORT")
    if path == "" { t.Fatal("missing report path") }
    raw, err := json.MarshalIndent(report, "", "  "); if err != nil { t.Fatal(err) }
    if err := os.WriteFile(path, append(raw, '\n'), 0644); err != nil { t.Fatal(err) }
}
