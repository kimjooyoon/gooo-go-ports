package main

import (
    "encoding/json"
    "os"
    "runtime"
    "strconv"
    "testing"
    "unicode"
)

// The generated composition remains unchanged in its own file. Only the oracle,
// exhaustive loop, explicit int64 extension contract and reporting live here.
func TestCaseMappingExhaustive(t *testing.T) {
    if runtime.Version() != "go1.27.3" || unicode.Version != "17.0.0" || strconv.IntSize != 64 {
        t.Fatal("wrong pinned Go / Unicode oracle or integer width")
    }
    cases := []int64{ {{range .ValidCases}}{{.}},{{end}} }
    outside := []int64{ {{range .OutsideUnicodeInt32Inputs}}{{.}},{{end}} }
    extensions := []int64{ {{range .OutsideInt32ExtensionInputs}}{{.}},{{end}} }
    invalidCases := []int64{ {{range .InvalidCases}}{{.}},{{end}} }
    invalidRunes := []int64{ {{range .InvalidCaseOriginalRuneInputs}}{{.}},{{end}} }
    actual := func(c, x int64) int64 { return {{.Function}}({{if .Binary}}c, {{end}}x) }
    oracle := func(c, x int64) int64 { return int64(unicode.{{.Operation}}({{if .Binary}}int(c), {{end}}rune(x))) }
    unicodeInputs, unicodeComparisons, outsideComparisons, extensionComparisons := 0, 0, 0, 0
    for x := int64(0); x <= int64(unicode.MaxRune); x++ {
        unicodeInputs++
        for _, c := range cases {
            want := oracle(c, x)
            if got := actual(c, x); got != want {
                t.Fatalf("{{.Operation}} case=%d rune=%d: got %d, want %d", c, x, got, want)
            }
            unicodeComparisons++
        }
    }
    for _, x := range outside {
        if x < -2147483648 || x > 2147483647 || (x >= 0 && x <= unicode.MaxRune) {
            t.Fatalf("invalid original int32 reference input %d", x)
        }
        for _, c := range cases {
            want := oracle(c, x)
            if got := actual(c, x); got != want {
                t.Fatalf("original int32 {{.Operation}} case=%d rune=%d: got %d, want %d", c, x, got, want)
            }
            outsideComparisons++
        }
    }
    for _, x := range extensions {
        if x >= -2147483648 && x <= 2147483647 { t.Fatalf("extension in original rune domain: %d", x) }
        for _, c := range cases {
            // No conversion to rune: these are explicitly identity extensions.
            if got := actual(c, x); got != x {
                t.Fatalf("int64 extension {{.Operation}} case=%d rune=%d: got %d, want original input", c, x, got)
            }
            extensionComparisons++
        }
    }
    invalidOriginalComparisons, invalidExtensionComparisons := 0, 0
    for _, c := range invalidCases {
        if c >= 0 && c <= 2 { t.Fatalf("valid case in invalid-case roster: %d", c) }
        for _, x := range invalidRunes {
            if x < -2147483648 || x > 2147483647 { t.Fatalf("invalid original rune: %d", x) }
            // int retains the exact int64 case on the pinned 64-bit toolchain.
            want := int64(unicode.To(int(c), rune(x)))
            if want != int64(unicode.ReplacementChar) { t.Fatal("invalid-case oracle differs") }
            if got := actual(c, x); got != want {
                t.Fatalf("invalid case original rune {{.Operation}} case=%d rune=%d: got %d, want %d", c, x, got, want)
            }
            invalidOriginalComparisons++
        }
        for _, x := range extensions {
            // Invalid case takes precedence over the identity extension. This
            // is not a Go-rune oracle comparison and must be counted separately.
            if got := actual(c, x); got != int64(unicode.ReplacementChar) {
                t.Fatalf("invalid case int64 extension {{.Operation}} case=%d rune=%d: got %d, want U+FFFD", c, x, got)
            }
            invalidExtensionComparisons++
        }
    }
    report := map[string]any{
        "schema": "gooo/case-mapping-exhaustive-native/v1",
        "go_version": runtime.Version(), "unicode_version": unicode.Version,
        "operation": "{{.Operation}}", "activity_id": "{{.ActivityID}}",
        "input_entity_ids": []string{ {{range .InputEntityIDs}}"{{.}}",{{end}} },
        "output_entity_id": "{{.OutputEntityID}}", "generated_sha256": "{{.GeneratedSHA256}}",
        "valid_cases": cases, "unicode_range_inputs": unicodeInputs, "unicode_range_comparisons": unicodeComparisons,
        "outside_unicode_int32_inputs": outside, "outside_unicode_int32_comparisons": outsideComparisons,
        "outside_int32_extension_inputs": extensions, "outside_int32_extension_comparisons": extensionComparisons,
        "invalid_cases": invalidCases, "invalid_case_original_rune_inputs": invalidRunes,
        "invalid_case_original_rune_comparisons": invalidOriginalComparisons,
        "invalid_case_extension_comparisons": invalidExtensionComparisons, "mismatches": 0,
    }
    path := os.Getenv("GOOO_CASE_MAPPING_REPORT")
    if path == "" { t.Fatal("missing report path") }
    raw, err := json.MarshalIndent(report, "", "  "); if err != nil { t.Fatal(err) }
    if err := os.WriteFile(path, append(raw, '\n'), 0644); err != nil { t.Fatal(err) }
}
