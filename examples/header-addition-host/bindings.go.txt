// Explicit typed ABI binding to the byte-preserved freshly generated source.
// This file only maps fields; it contains no policy predicates or fallback.
package main

type assessorFacts struct {
	ContentLengthOccurrences      int64  `json:"content_length_occurrences"`
	ObservedBodyBytes             int64  `json:"observed_body_bytes"`
	MaximumPayloadBytes           int64  `json:"maximum_payload_bytes"`
	TransferEncodingPresent       bool   `json:"transfer_encoding_present"`
	BodyObservationBoundToRequest bool   `json:"body_observation_bound_to_request"`
	CurrentAccess                 bool   `json:"current_access"`
	CandidateNameOccurrences      int64  `json:"candidate_name_occurrences"`
	HeaderSetBoundToRequest       bool   `json:"header_set_bound_to_request"`
	PolicyRevisionCurrent         bool   `json:"policy_revision_current"`
	AllowedHeaderName             string `json:"allowed_header_name"`
	ExpectedRequestRevision       int64  `json:"expected_request_revision"`
	ObservedRequestRevision       int64  `json:"observed_request_revision"`
}

type Decision struct {
	CandidateName       string `json:"candidate_name"`
	CandidateValue      string `json:"candidate_value"`
	Status              string `json:"status"`
	Reason              string `json:"reason"`
	PayloadEligible     bool   `json:"payload_eligible"`
	PayloadReason       int64  `json:"payload_reason"`
	HeaderNameValid     bool   `json:"header_name_valid"`
	HeaderValueValid    bool   `json:"header_value_valid"`
	NameInvalidOffset   int64  `json:"name_invalid_offset"`
	NameInvalidOctet    int64  `json:"name_invalid_octet"`
	ValueInvalidOffset  int64  `json:"value_invalid_offset"`
	ValueInvalidOctet   int64  `json:"value_invalid_octet"`
	ValueHasObsText     bool   `json:"value_has_obs_text"`
	ValueEdgeWhitespace bool   `json:"value_edge_whitespace"`
	RequestRevision     int64  `json:"request_revision"`
	Eligible            bool   `json:"eligible"`
}
